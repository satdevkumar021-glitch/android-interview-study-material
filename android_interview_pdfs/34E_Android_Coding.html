2. Parallel API Calls

Fetch user + orders simultaneously

data class User(val id: String, val name: String)
data class Order(val orderId: String, val total: Double)
data class ProfilePage(val user: User?, val orders: List<Order>)

// ✅ Parallel: both requests start simultaneously
suspend fun loadProfilePage(
    userId: String,
    api: ProfileApi
): ProfilePage = coroutineScope {
    // Launch both concurrently; runCatching per-request for independent failure
    val userDeferred = async(Dispatchers.IO) {
        runCatching { api.getUser(userId) }
    }
    val ordersDeferred = async(Dispatchers.IO) {
        runCatching { api.getOrders(userId) }
    }

    val user   = userDeferred.await().getOrNull()
    val orders = ordersDeferred.await().getOrDefault(emptyList())

    ProfilePage(user, orders)
}

// In ViewModel:
class ProfileViewModel @Inject constructor(private val repo: ProfileRepository) : ViewModel() {
    private val _uiState = MutableStateFlow<ProfileUiState>(ProfileUiState.Loading)
    val uiState = _uiState.asStateFlow()

    fun loadProfile(userId: String) {
        viewModelScope.launch {
            _uiState.value = ProfileUiState.Loading
            _uiState.value = runCatching { repo.loadProfile(userId) }
                .fold(
                    onSuccess = { ProfileUiState.Success(it) },
                    onFailure = { ProfileUiState.Error(it.message ?: "Unknown error") }
                )
        }
    }
}

3. Sequential API Calls

Fetch product, then fetch recommendations by category

data class Product(val id: String, val name: String, val category: String)
data class Recommendation(val productId: String)
data class ProductDetail(val product: Product, val recs: List<Recommendation>)

// Sequential: step 2 depends on step 1 result
suspend fun loadProductDetail(
    productId: String,
    api: ProductApi
): ProductDetail = withContext(Dispatchers.IO) {
    // Step 1: must complete before step 2
    val product = api.getProduct(productId)   // throws on failure

    // Step 2: uses product.category from step 1
    val recs = runCatching {
        api.getRecommendations(product.category)
    }.getOrDefault(emptyList())   // optional — show product even if recs fail

    ProductDetail(product, recs)
}

// Multi-step with explicit error handling
suspend fun checkoutFlow(cartId: String, api: CheckoutApi): OrderConfirmation {
    return withContext(Dispatchers.IO) {
        val cart        = api.validateCart(cartId)        // throws → propagates
        val pricing     = api.calculatePricing(cart)      // uses cart
        val paymentAuth = api.authorizePayment(pricing)   // uses pricing
        api.placeOrder(cart, paymentAuth)                  // final step
    }
}
// Any step throwing cancels the whole coroutine — exception propagates to caller

4. Flow Transformations: Search Debounce

Classic pattern: user types in a search field → debounce 300ms → API call → map to UI state → handle errors → display results.

// Complete search debounce implementation
class SearchViewModel @Inject constructor(
    private val repository: SearchRepository
) : ViewModel() {

    // StateFlow for search query input
    private val _searchQuery = MutableStateFlow("")

    // Results flow: debounced, transformed, error-handled
    val searchResults: StateFlow<SearchUiState> = _searchQuery
        .debounce(300L)           // wait 300ms of inactivity before emitting
        .distinctUntilChanged()    // skip if query didn't change
        .filter { it.length >= 2 } // min 2 chars to avoid trivial searches
        .flatMapLatest { query ->   // cancel previous search if new query arrives
            flow {
                emit(SearchUiState.Loading)
                val results = repository.search(query)
                emit(SearchUiState.Success(results))
            }.catch { e ->
                emit(SearchUiState.Error(e.message ?: "Search failed"))
            }
        }
        .stateIn(
            scope = viewModelScope,
            started = SharingStarted.WhileSubscribed(5000),
            initialValue = SearchUiState.Idle
        )

    fun onSearchQueryChanged(query: String) {
        _searchQuery.value = query
    }
}

// UI State
sealed class SearchUiState {
    object Idle : SearchUiState()
    object Loading : SearchUiState()
    data class Success(val results: List<SearchResult>) : SearchUiState()
    data class Error(val message: String) : SearchUiState()
}

// Compose UI:
// TextField(value = query, onValueChange = { viewModel.onSearchQueryChanged(it) })
// when (uiState) { is Loading → CircularProgressIndicator(); is Success → ... }
flatMapLatest vs flatMapMerge
flatMapLatest: cancels the previous inner Flow when a new value arrives — perfect for search (cancel inflight API call on new keystroke). flatMapMerge: runs all inner Flows concurrently — use for independent operations where you want all results. flatMapConcat: runs sequentially.

5. LRU Cache Implementation

Used in image caching explanation — implement a fixed-size cache that evicts the least recently used entry when full.

// LRU Cache using LinkedHashMap
// LinkedHashMap maintains insertion order by default
// accessOrder=true makes it maintain ACCESS order (most recent last)
class LruCache<K, V>(private val maxSize: Int) {

    private val cache = object : LinkedHashMap<K, V>(
        // initialCapacity, loadFactor, accessOrder=true (orders by access, not insertion)
        maxSize, 0.75f, /* accessOrder = */ true
    ) {
        override fun removeEldestEntry(eldest: Map.Entry<K, V>): Boolean {
            return size > maxSize   // evict oldest when over capacity
        }
    }

    fun get(key: K): V? = cache[key]   // also updates access order

    fun put(key: K, value: V) { cache[key] = value }

    fun remove(key: K) = cache.remove(key)

    val size get() = cache.size

    override fun toString() = cache.toString()
}

// Test
val lru = LruCache<Int, String>(3)
lru.put(1, "one")
lru.put(2, "two")
lru.put(3, "three")
lru.get(1)       // access key 1 — moves to "most recently used"
lru.put(4, "four")  // evicts key 2 (least recently used)
println(lru.get(2))  // null — evicted
println(lru.get(1))  // "one" — still in cache

// Thread-safe wrapper if needed:
class ThreadSafeLruCache<K, V>(maxSize: Int) {
    private val cache = LruCache<K, V>(maxSize)
    private val lock = Any()
    fun get(key: K): V? = synchronized(lock) { cache.get(key) }
    fun put(key: K, value: V) = synchronized(lock) { cache.put(key, value) }
}

6. Token Refresh Interceptor (Complete Implementation)

The complete production-ready implementation of concurrent token refresh prevention with Mutex + OkHttp Authenticator.

// ===== TokenManager: handles token storage and refresh =====
class TokenManager @Inject constructor(
    private val authApi: AuthApi,
    private val tokenStorage: TokenStorage
) {
    private val mutex = Mutex()

    suspend fun getValidToken(): String {
        // Fast path: return cached token if still valid
        tokenStorage.getToken()
            ?.let { if (!isExpired(it)) return it }

        // Slow path: refresh needed. Only ONE coroutine refreshes at a time.
        mutex.withLock {
            // Double-check: another coroutine may have refreshed while we waited for lock
            tokenStorage.getToken()
                ?.let { if (!isExpired(it)) return it }

            // Actually refresh
            val newTokens = authApi.refreshToken(tokenStorage.getRefreshToken()!!)
            tokenStorage.saveToken(newTokens.accessToken)
            tokenStorage.saveRefreshToken(newTokens.refreshToken)
            return newTokens.accessToken
        }
    }

    private fun isExpired(token: String): Boolean {
        // Parse JWT expiry from payload, compare with System.currentTimeMillis()
        return try {
            val payload = token.split(".")[1]
            val decoded = Base64.decode(payload, Base64.URL_SAFE)
            val json = JSONObject(String(decoded))
            val exp = json.getLong("exp") * 1000L
            System.currentTimeMillis() >= exp - 30_000L   // 30s buffer
        } catch (e: Exception) { true }   // if parse fails, assume expired
    }
}

// ===== OkHttp Authenticator: handles 401 responses =====
class TokenAuthenticator @Inject constructor(
    private val tokenManager: TokenManager
) : Authenticator {

    override fun authenticate(route: Route?, response: Response): Request? {
        // Stop retry loop after 1 refresh attempt
        if (response.responseCount() >= 2) return null

        val newToken = runBlocking {
            tokenManager.getValidToken()   // Mutex inside prevents concurrent refreshes
        }

        return response.request.newBuilder()
            .header("Authorization", "Bearer $newToken")
            .build()
    }

    private fun Response.responseCount(): Int {
        var count = 1; var prior = priorResponse
        while (prior != null) { count++; prior = prior.priorResponse }
        return count
    }
}

// ===== Add to OkHttp client =====
val okHttpClient = OkHttpClient.Builder()
    .authenticator(tokenAuthenticator)
    .addInterceptor { chain ->
        // Add current token to every request
        val token = runBlocking { tokenManager.getValidToken() }
        chain.proceed(
            chain.request().newBuilder()
                .header("Authorization", "Bearer $token")
                .build()
        )
    }
    .build()

7. Interview Framework
Android Coding Interview Template
When given an Android coding challenge: (1) State the coroutine scope you'd use (viewModelScope/coroutineScope). (2) State the dispatcher (IO for network/DB). (3) Handle errors with runCatching or try/catch. (4) Always emit Loading state before async work. (5) Mention cancellation — what gets cancelled automatically vs what needs manual cleanup.

Key Points to Mention

  • Parallel calls: coroutineScope { async + async }; if one fails independently, use runCatching per async
  • Search debounce: debounce(300) → distinctUntilChanged → flatMapLatest (cancels previous on new input)
  • LRU: LinkedHashMap with accessOrder=true + removeEldestEntry override; thread-safe with synchronized
  • Token refresh: Mutex + double-check pattern prevents duplicate refreshes; OkHttp Authenticator handles 401

9. Practice Questions
Why use flatMapLatest for a search Flow instead of flatMapMerge?
Medium▼
flatMapLatest cancels the previous inner coroutine/Flow when a new value arrives. For search: if the user types "an" then immediately "and", the "an" API call is cancelled and only "and" is executed. This prevents stale results from "an" arriving after "and" results, and saves bandwidth. flatMapMerge would run both concurrently — "an" results might arrive after "and" results and overwrite them, showing wrong data. flatMapConcat would queue them — search for "an" completes before "and" starts, causing unnecessary delay.
Why is the double-check pattern inside the Mutex necessary for token refresh?
Hard▼
5 coroutines detect the token is expired and race to acquire the Mutex. Coroutine #1 acquires it, refreshes the token, saves it, and releases the lock. Coroutines #2–5 now acquire the lock one by one. Without double-check: each of #2–5 would refresh again (wasting 4 network calls, potentially invalidating the previous refresh). With double-check: #2 acquires lock, re-checks the cached token — it's now valid (refreshed by #1), returns immediately without refreshing. Same for #3, #4, #5.
What does SharingStarted.WhileSubscribed(5000) do in stateIn?
Medium▼
WhileSubscribed(5000) keeps the upstream Flow active while there are active subscribers, and for 5000ms after the last subscriber unsubscribes. This handles screen rotation: when the Fragment's lifecycle reaches STOPPED and unsubscribes, the Flow keeps running for 5 seconds. If the user rotates quickly (within 5s), the new subscriber gets the same running Flow — no restart, no re-fetch. After 5s of no subscribers (user navigated away), the Flow stops collecting and cancels upstream.
How does LinkedHashMap with accessOrder=true implement LRU behavior?
Medium▼
With accessOrder=true, every get() or put() operation moves that entry to the tail (most recently used end) of the internal doubly-linked list. The head is always the least recently used entry. removeEldestEntry is called after each put() — returning true when size > maxSize removes the head (LRU entry) automatically. This gives O(1) get, put, and eviction. The LinkedHashMap maintains both a HashMap (O(1) lookup) and a doubly-linked list (access order) internally.
When would you use coroutineScope vs supervisorScope for parallel API calls?
Hard▼
coroutineScope: if any child fails (throws), ALL siblings are cancelled and the exception propagates. Use when all results are required — failure in one means the whole operation fails. supervisorScope: children fail independently — one failure doesn't cancel others. Use when requests are independent and you want partial results. For the profile page: if user is required but orders are optional, use supervisorScope with runCatching on the orders async. For a multi-step checkout where each step is required, use coroutineScope.
Why is runBlocking used in OkHttp Authenticator and is it safe?
Hard▼
OkHttp interceptors and Authenticator run on OkHttp's thread pool — they are synchronous (blocking) APIs. There's no way to use suspend functions directly. runBlocking bridges the synchronous OkHttp world with the coroutine-based TokenManager. It's safe here because: OkHttp's thread pool is NOT the main thread, so blocking it is acceptable (no ANR risk). The thread is blocked for only the duration of the token refresh (~100ms). If you never call the Authenticator on the main thread (you shouldn't be), runBlocking is safe.
Implement a simple debounce function using coroutines (without using the debounce operator).
Hard▼
fun <T> debounce(delayMs: Long, scope: CoroutineScope, action: suspend (T) -> Unit): (T) -> Unit { var debounceJob: Job? = null; return { value: T -> debounceJob?.cancel(); debounceJob = scope.launch { delay(delayMs); action(value) } } }. Usage: val debouncedSearch = debounce(300L, viewModelScope) { query -> performSearch(query) }; textField.onTextChange { debouncedSearch(it) }. Each call cancels the previous pending job and schedules a new one. After 300ms of inactivity, the action fires.
How does the catch operator on a Flow differ from a try/catch around collect?
Medium▼
catch {} is a Flow operator that only catches exceptions from upstream (emitters above it in the chain). It can emit recovery values with emit(). try/catch around collect{} catches exceptions from both upstream AND the collector (your UI code inside collect). Using catch in the repository/ViewModel layer keeps error handling upstream and allows the Flow to emit an error state instead of terminating. Place catch before terminal operators like collect or stateIn. If an exception occurs in collect{} (the UI code), catch won't catch it — use try/catch there.
How would you make the LRU cache coroutine-safe instead of using synchronized?
Hard▼
Replace synchronized(lock) with Mutex: class CoroutineLruCache<K, V>(maxSize: Int) { private val mutex = Mutex(); private val cache = LruCache<K, V>(maxSize); suspend fun get(key: K): V? = mutex.withLock { cache.get(key) }; suspend fun put(key: K, value: V) = mutex.withLock { cache.put(key, value) } }. Mutex is coroutine-aware — when withLock waits, it suspends the coroutine instead of blocking the thread. This is more efficient in a coroutine-heavy app. synchronized blocks the calling thread; Mutex suspends the coroutine and allows the thread to serve other coroutines while waiting.
Write a Flow that retries a failed network call with exponential backoff.
Hard▼
fun fetchWithRetry(maxRetries: Int = 3): Flow<Result> = flow { emit(api.fetchData()) }.retryWhen { cause, attempt -> if (attempt < maxRetries && cause is IOException) { delay(2.0.pow(attempt.toInt()).toLong() * 1000L); true } else false }. The retryWhen operator: cause is the exception, attempt is 0-based retry count. delay grows: 1s, 2s, 4s. Returns true to retry, false to propagate. Add a jitter: + Random.nextLong(0, 500) to avoid thundering herd. Only retry on transient errors (IOException) not on server errors (HttpException with 4xx).
