7E — Compose UI

Tier 1 Row · Column · Box · LazyColumn · LazyRow · Scaffold · Material 3 · Modifiers · Custom Layouts

1 · Overview

Jetpack Compose's UI toolkit provides a set of composable layout functions that replace XML views entirely. Layouts are described in Kotlin as composable function calls, and the Compose runtime handles measuring, placing, and drawing efficiently.

Core principle: Every composable is a function that emits UI nodes into the composition tree. Layout composables (Row, Column, Box) control how child nodes are measured and positioned. Modifier adds decoration and behaviour without subclassing.

Modifiers

Modifiers are applied left-to-right, each wrapping the next. Order matters — padding before background fills the padded area; background before padding fills outside the padding.

// Order matters: size → background → padding → clickable
Box(
    Modifier
        .fillMaxWidth()
        .height(56.dp)
        .background(MaterialTheme.colorScheme.primary)
        .padding(horizontal = 16.dp)
        .clickable { onTap() }
)

Scaffold — Material 3

@Composable
fun BankingScreen(
    viewModel: BankingViewModel,
    onNavigate: (Route) -> Unit
) {
    Scaffold(
        topBar = {
            TopAppBar(
                title = { Text("Accounts") },
                navigationIcon = { IconButton(onClick = { onNavigate(Route.Back) }) {
                    Icon(Icons.Default.ArrowBack, "Back")
                }}
            )
        },
        floatingActionButton = {
            FloatingActionButton(onClick = { onNavigate(Route.NewTransfer) }) {
                Icon(Icons.Default.Add, "New transfer")
            }
        }
    ) { paddingValues ->
        // paddingValues accounts for TopAppBar + FAB insets
        AccountsList(
            modifier = Modifier.padding(paddingValues),
            viewModel = viewModel
        )
    }
}

LazyColumn with Keys

@Composable
fun TransactionList(transactions: List<Transaction>) {
    LazyColumn(
        contentPadding = PaddingValues(16.dp),
        verticalArrangement = Arrangement.spacedBy(8.dp)
    ) {
        items(
            items = transactions,
            key = { it.id }   // ✅ stable key — preserves scroll pos, avoids full recompose
        ) { tx ->
            TransactionItem(transaction = tx)
        }
    }
}

Custom Layout

// Custom layout: vertical list where each item has a dot timeline connector
@Composable
fun TimelineLayout(content: @Composable () -> Unit) {
    Layout(content = content) { measurables, constraints ->
        val placeables = measurables.map { it.measure(constraints) }
        val height = placeables.sumOf { it.height }
        layout(width = constraints.maxWidth, height = height) {
            var y = 0
            placeables.forEach { placeable ->
                placeable.placeRelative(x = 0, y = y)
                y += placeable.height
            }
        }
    }
}

4 · Production Example — E-Commerce Product Screen

@Composable
fun ProductDetailScreen(
    product: Product,
    onAddToCart: () -> Unit,
    onBack: () -> Unit
) {
    Scaffold(
        topBar = {
            TopAppBar(
                title = { Text(product.name, maxLines = 1, overflow = TextOverflow.Ellipsis) },
                navigationIcon = {
                    IconButton(onClick = onBack) {
                        Icon(Icons.AutoMirrored.Default.ArrowBack, "Back")
                    }
                }
            )
        },
        bottomBar = {
            AddToCartBar(price = product.price, onAddToCart = onAddToCart)
        }
    ) { padding ->
        LazyColumn(
            modifier = Modifier.padding(padding),
            contentPadding = PaddingValues(bottom = 16.dp)
        ) {
            item(key = "hero") {
                AsyncImage(
                    model = product.imageUrl,
                    contentDescription = product.name,
                    modifier = Modifier.fillMaxWidth().height(280.dp),
                    contentScale = ContentScale.Crop
                )
            }
            item(key = "details") {
                Column(Modifier.padding(16.dp)) {
                    Text(product.name, style = MaterialTheme.typography.headlineSmall)
                    Text(product.description, style = MaterialTheme.typography.bodyMedium,
                        color = MaterialTheme.colorScheme.onSurfaceVariant)
                }
            }
            items(product.reviews, key = { it.id }) { review ->
                ReviewItem(review = review)
            }
        }
    }
}

@Composable
private fun AddToCartBar(price: Double, onAddToCart: () -> Unit) {
    Surface(shadowElevation = 8.dp) {
        Row(
            modifier = Modifier.fillMaxWidth().padding(16.dp),
            horizontalArrangement = Arrangement.SpaceBetween,
            verticalAlignment = Alignment.CenterVertically
        ) {
            Text("$$price", style = MaterialTheme.typography.headlineSmall)
            Button(onClick = onAddToCart) { Text("Add to Cart") }
        }
    }
}

6 · Pitfalls and Best Practices

  • LazyColumn without keys: Without key, Compose cannot identify item identity — scroll state lost, animations wrong, unnecessary recompositions.
  • Modifier order: .padding(16.dp).background(Color.Red) vs .background(Color.Red).padding(16.dp) produce different results. Background before padding = background under the padding area.
  • Missing padding(paddingValues) in Scaffold: Content overlaps TopAppBar / BottomBar if you don't apply the Scaffold's paddingValues to your content.
  • Column for long lists: Column measures all children eagerly — use LazyColumn for any list that might grow.
  • Modifier.clickable without ripple: Use Modifier.clickable { } (has built-in ripple) not custom pointerInput unless you need fine-grained gesture control.
  • Hard-coded dp in custom layouts: Use LocalDensity to convert px ↔ dp in custom layouts, not hard-coded pixel values.
Best practices: Always pass paddingValues in Scaffold. Always provide key in LazyColumn. Use Arrangement.spacedBy() for consistent gaps. Use fillMaxWidth() on top-level layout composables. Apply Modifier parameters in a consistent order: size → background → padding → border → click.

7 · Interview Answer Framework

30-Second Answer

"Compose UI is built around three layout primitives — Row (horizontal), Column (vertical), Box (stack) — plus lazy variants LazyColumn and LazyRow for scrollable lists. Scaffold provides the Material 3 chrome (TopAppBar, BottomBar, FAB, Snackbar). Modifiers decorate and add behaviour without subclassing, and their order matters. LazyColumn should always have a stable key per item for correct diffing and scroll preservation. Custom layouts use the Layout composable for full measurement control."

2-Minute Structured Answer

Layouts: Row/Column/Box for simple arrangements; LazyColumn/LazyRow for scrollable lists (analogous to RecyclerView but no Adapter boilerplate). Scaffold is the standard screen wrapper — always pass its paddingValues to the content to avoid overlap with TopAppBar/BottomBar.

Modifiers: Applied left-to-right; each wraps the previous. Order matters — size, then background, then padding, then click is the standard pattern. clickable has a built-in ripple; use it instead of custom gesture handlers unless necessary.

LazyColumn performance: Always provide stable keys — without them, Compose cannot tell which item moved vs was replaced, causing wrong animations and losing scroll state. Use contentPadding and Arrangement.spacedBy() for spacing.

Material 3: MaterialTheme provides colorScheme, typography, shapes. Use MaterialTheme.colorScheme.primary etc. rather than hard-coded colours — themes correctly in dark mode automatically.

Must-Mention Points

  1. Row/Column/Box are the three layout primitives; Lazy variants for scrollable lists.
  2. Modifier order matters (especially padding vs background).
  3. Scaffold for Material 3 screen chrome — always use paddingValues.
  4. LazyColumn keys — critical for correctness and performance.
  5. Custom Layout composable for non-standard placement.
  6. Material 3 theming via MaterialTheme (colorScheme, typography, shapes).

Senior-Level Insight

"The key architectural insight with Compose layouts is that there are no XML inflation or view recycling concerns — LazyColumn handles recycling automatically, and the Compose compiler tracks which items need recomposition via stable keys. The main performance lever is stability: if your list item data class is @Stable or @Immutable and you provide a key, Compose will skip recomposing unchanged items entirely. This is more powerful than DiffUtil in RecyclerView because the skip decision happens per-composable rather than per-list-item."

9 · Interview Questions (Self-Check)

Basic Name the three core Compose layout primitives and their XML equivalents.
Row (LinearLayout horizontal), Column (LinearLayout vertical), Box (FrameLayout). LazyColumn and LazyRow are the lazy/scrollable equivalents of RecyclerView.
Basic Why does Modifier order matter in Compose?
Each Modifier wraps the previous one. Modifiers closer to the composable affect the inner content; modifiers further out affect the outer boundary. Example: Modifier.padding(16dp).background(Red) applies background only to the inner content area. Modifier.background(Red).padding(16dp) applies background to the full area including the padding.
Basic What is Scaffold and why must you use its paddingValues?
Scaffold is the Material 3 screen scaffold that provides slots for TopAppBar, BottomNavigationBar, FAB, Snackbar, and DrawerLayout. The paddingValues it passes to the content lambda represent the insets consumed by these components (TopAppBar height, BottomBar height). If you don't apply paddingValues to your content, it will be drawn behind the TopAppBar and BottomBar.
Intermediate Why should LazyColumn items always have a key? What happens without it?
Without a key, Compose uses positional identity (index 0, 1, 2...). When items are inserted/removed/reordered, Compose recomposes all subsequent items because their positions changed. With a stable key (item ID), Compose can identify which items moved vs changed, skip recomposing unchanged items, apply correct animations, and preserve scroll state across recompositions. Key must be unique and stable (no random/UUID generated each time).
Intermediate How do you add equal spacing between LazyColumn items?
Use verticalArrangement = Arrangement.spacedBy(8.dp) in the LazyColumn parameters. Don't add padding to individual item composables — that creates asymmetric spacing (double gap between items, single gap at edges). For outer padding (top/bottom of the whole list), use contentPadding = PaddingValues(vertical = 16.dp).
Intermediate When would you use ConstraintLayout in Compose?
When a layout requires complex sibling constraints that would otherwise need deeply nested Row/Column combinations. ConstraintLayout computes its layout in a single pass regardless of constraint complexity, whereas deeply nested Row/Column can cause multiple measurement passes (though Compose's intrinsic measurement largely mitigates this). Use it for: flexible responsive layouts, baseline alignment between sibling elements, and adaptive layouts that change significantly at different screen sizes.
Intermediate How do you create a custom layout in Compose?
Use the Layout composable: provide a content lambda (child composables) and a measurePolicy lambda. The measurePolicy receives measurables (children to measure) and constraints (available space). Call measurable.measure(constraints) to get a Placeable. Call layout(width, height) { } to set the layout size and position children using placeable.placeRelative(x, y).
Senior/Scenario [Scenario] A LazyColumn with 1,000 items is janky on scroll. How do you diagnose and optimize it?
Diagnosis: Use Android Studio Layout Inspector in Compose mode to see recomposition counts. Check Compose compiler metrics for unstable composables. Use systrace/Perfetto to find dropped frames.

Fixes:
1. Add stable keys to items — prevents wrong recompositions
2. Make item data class @Stable or @Immutable — allows Compose to skip recomposition
3. Use remember{} for expensive computations inside item composable
4. Avoid lambdas that capture state inside item composable (use callbacks passed as parameters)
5. Use contentType = { it.type } in items{} if you have heterogeneous item types — improves view recycling
6. Avoid Modifier.graphicsLayer inside items unless needed for animation
7. Check for nested LazyColumns (not allowed — causes measurement exceptions)
Senior/Scenario [Coding] Implement a horizontally scrollable chip row that wraps to next line if chips don't fit (flow layout). 
// FlowRow is available in Compose Foundation (experimental in 1.4, stable in 1.5+)
@Composable
fun FilterChipRow(
    filters: List<String>,
    selected: Set<String>,
    onToggle: (String) -> Unit
) {
    FlowRow(
        horizontalArrangement = Arrangement.spacedBy(8.dp),
        verticalArrangement = Arrangement.spacedBy(8.dp),
        modifier = Modifier.fillMaxWidth().padding(horizontal = 16.dp)
    ) {
        filters.forEach { filter ->
            FilterChip(
                selected = filter in selected,
                onClick = { onToggle(filter) },
                label = { Text(filter) }
            )
        }
    }
}
// FlowRow automatically wraps chips to next line when horizontal space is exhausted
Senior/Scenario [Debug] Content in a Scaffold is hidden behind the TopAppBar. What's wrong?
The paddingValues provided by Scaffold's content lambda is not being applied to the content. Scaffold calculates insets from TopAppBar, BottomBar, and FAB and passes them as PaddingValues to the content slot. If you don't apply them (via Modifier.padding(paddingValues)), your content starts at y=0, which is behind the TopAppBar.
Fix:
Scaffold(topBar = { TopAppBar(...) }) { paddingValues ->
    LazyColumn(modifier = Modifier.padding(paddingValues)) { // ✅
        ...
    }
}
Also verify you're not applying contentPadding to the LazyColumn that conflicts with paddingValues.
