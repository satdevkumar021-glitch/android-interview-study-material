7F — Compose Navigation

Tier 1 NavController · NavHost · Arguments · Nested Navigation · Deep Links · Back Stack

1 · Overview

Navigation Compose is the Jetpack Navigation library adapted for Compose. It manages a back stack of Composable destinations, enabling type-safe navigation, argument passing, deep links, and nested navigation graphs — all without Activity or Fragment boilerplate.

Core components:
  • NavController — manages navigation state and back stack
  • NavHost — the Composable that renders the current destination
  • NavGraph — the set of destinations and connections, defined in code

2 · How It Works

Basic Setup

MainActivity (single Activity) └── NavHost ├── composable("home") → HomeScreen() ├── composable("product/{id}") → ProductDetailScreen() ├── composable("cart") → CartScreen() └── navigation("auth") → nested AuthGraph ├── composable("login") → LoginScreen() └── composable("register") → RegisterScreen()
@Composable
fun AppNavHost(navController: NavHostController = rememberNavController()) {
    NavHost(
        navController = navController,
        startDestination = "home"
    ) {
        composable("home") {
            HomeScreen(onNavigateToProduct = { id ->
                navController.navigate("product/$id")
            })
        }
        composable(
            route = "product/{productId}",
            arguments = listOf(navArgument("productId") { type = NavType.StringType })
        ) { backStackEntry ->
            val productId = backStackEntry.arguments?.getString("productId")
            ProductDetailScreen(productId = productId!!)
        }
    }
}

Type-Safe Navigation (Navigation 2.8+ with Kotlin Serialization)

// Define routes as sealed class or @Serializable objects (Navigation 2.8+)
@Serializable object HomeRoute
@Serializable data class ProductRoute(val productId: String)
@Serializable object CartRoute

NavHost(navController, startDestination = HomeRoute) {
    composable<HomeRoute> { HomeScreen(...) }
    composable<ProductRoute> { backStackEntry ->
        val route: ProductRoute = backStackEntry.toRoute()
        ProductDetailScreen(productId = route.productId)
    }
}

// Navigate with type safety
navController.navigate(ProductRoute(productId = "abc123"))

Back Stack Management

// Navigate and clear login from back stack (logout flow)
navController.navigate(HomeRoute) {
    popUpTo(LoginRoute::class) { inclusive = true }  // remove Login from stack
}

// launchSingleTop — don't stack duplicates
navController.navigate(HomeRoute) {
    popUpTo(HomeRoute::class) { inclusive = true }
    launchSingleTop = true
}

// Navigate up (Back)
navController.navigateUp()    // preferred over popBackStack()

Nested Navigation Graphs

NavHost(navController, startDestination = "main") {
    // Nested auth graph — separate back stack scope
    navigation(startDestination = "login", route = "auth") {
        composable("login") { LoginScreen(...) }
        composable("register") { RegisterScreen(...) }
    }
    navigation(startDestination = "home", route = "main") {
        composable("home") { HomeScreen(...) }
        composable("cart") { CartScreen(...) }
    }
}

// Scoped ViewModel to nested graph — shared between screens in that graph
val authViewModel: AuthViewModel = viewModel(
    viewModelStoreOwner = remember(navController) {
        navController.getBackStackEntry("auth")
    }
)

Deep Links

composable(
    route = "order/{orderId}",
    deepLinks = listOf(navDeepLink {
        uriPattern = "https://mybank.com/orders/{orderId}"
    })
) { backStackEntry ->
    val orderId = backStackEntry.arguments?.getString("orderId")
    OrderDetailScreen(orderId!!)
}

4 · Production Example — Banking App with Tabs + Auth Flow

// Sealed route hierarchy
@Serializable sealed class Route {
    @Serializable object Login : Route()
    @Serializable object Home : Route()
    @Serializable data class TransactionDetail(val txId: String) : Route()
    @Serializable object Transfer : Route()
}

@Composable
fun AppNavHost() {
    val navController = rememberNavController()
    val authState by authViewModel.authState.collectAsStateWithLifecycle()

    // Navigate to login when auth state changes to logged out
    LaunchedEffect(authState) {
        if (authState is AuthState.LoggedOut) {
            navController.navigate(Route.Login) {
                popUpTo(0) { inclusive = true }   // clear entire back stack
            }
        }
    }

    NavHost(
        navController = navController,
        startDestination = if (authState is AuthState.LoggedIn) Route.Home else Route.Login
    ) {
        composable<Route.Login> {
            LoginScreen(onLoginSuccess = {
                navController.navigate(Route.Home) {
                    popUpTo(Route.Login::class) { inclusive = true }
                }
            })
        }
        composable<Route.Home> {
            HomeScreen(onTransactionClick = { txId ->
                navController.navigate(Route.TransactionDetail(txId))
            })
        }
        composable<Route.TransactionDetail> { entry ->
            val route: Route.TransactionDetail = entry.toRoute()
            TransactionDetailScreen(txId = route.txId, onBack = navController::navigateUp)
        }
    }
}

Bottom Navigation with State Preservation

@Composable
fun MainScreen() {
    val navController = rememberNavController()
    Scaffold(
        bottomBar = {
            NavigationBar {
                val currentDest by navController.currentBackStackEntryAsState()
                tabs.forEach { tab ->
                    NavigationBarItem(
                        selected = currentDest?.destination?.hierarchy?.any { it.route == tab.route } == true,
                        onClick = {
                            navController.navigate(tab.route) {
                                popUpTo(navController.graph.findStartDestination().id) { saveState = true }
                                launchSingleTop = true
                                restoreState = true   // restore tab state on re-select
                            }
                        },
                        icon = { Icon(tab.icon, tab.label) },
                        label = { Text(tab.label) }
                    )
                }
            }
        }
    ) { padding ->
        NavHost(navController, "home", Modifier.padding(padding)) {
            navigation(startDestination = "accounts", route = "home") { /* ... */ }
            navigation(startDestination = "transactions", route = "history") { /* ... */ }
            navigation(startDestination = "profile", route = "settings") { /* ... */ }
        }
    }
}

6 · Pitfalls and Best Practices

  • NavController not at composition top: Create NavController with rememberNavController() at the top of the composition tree and pass it down. Don't create it inside individual screens.
  • Passing complex objects as nav args: Nav args are strings (Bundle-backed). Serialize only IDs; load full data in the destination ViewModel. Navigation 2.8 typed routes support basic types only via kotlinx.serialization.
  • Missing saveState/restoreState in tab navigation: Without saveState = true / restoreState = true, switching tabs recreates the entire tab screen. This loses scroll position and state.
  • Navigating in LaunchedEffect without correct keys: Always use a stable key (e.g., event or state value) in LaunchedEffect to avoid re-triggering navigation on recomposition.
  • Back stack growth from bottom nav: Without popUpTo(startDestination), repeatedly tapping tab icons stacks destinations. Always use the bottom nav pattern with popUpTo + launchSingleTop.
  • ViewModel scope in nested graphs: Use navController.getBackStackEntry("graphRoute") as the ViewModelStoreOwner to scope a ViewModel to a graph, not to individual destinations.

7 · Interview Answer Framework

30-Second Answer

"Navigation Compose uses NavController to manage a back stack of composable destinations declared in a NavHost. Each destination is a composable{} block with a route string or @Serializable type-safe route. Arguments are typed with navArgument, deep links with navDeepLink. Nested navigation graphs scope back stacks and ViewModels. For bottom tabs, popUpTo + launchSingleTop + saveState/restoreState are the canonical pattern to avoid back stack growth and preserve tab state."

2-Minute Structured Answer

Core components: NavController (back stack state), NavHost (renders current destination), NavGraph (set of destinations defined in code).

Arguments: Typed via navArgument() — NavType.StringType, IntType, BoolType. Navigation 2.8+ supports @Serializable data class routes for full type safety with toRoute().

Back stack: popUpTo() removes destinations; inclusive=true removes the target itself. For logout: popUpTo(0) { inclusive=true } clears the entire graph. For tab nav: popUpTo(startDestination) { saveState=true } + launchSingleTop=true + restoreState=true.

Nested graphs: navigation{} blocks create sub-graphs with their own start destination. Useful for auth flow (scope AuthViewModel to the auth graph) and feature modules. ViewModel scoped to graph: viewModel(viewModelStoreOwner = navController.getBackStackEntry("graphRoute")).

Must-Mention Points

  1. NavController + NavHost + NavGraph are the three core components.
  2. Type-safe navigation with @Serializable routes (Navigation 2.8+).
  3. popUpTo + inclusive for clearing back stack (logout, auth flows).
  4. Bottom navigation: popUpTo + launchSingleTop + saveState + restoreState.
  5. Nested navigation graphs for scoped ViewModels and feature isolation.
  6. Deep links via navDeepLink{} in composable declaration.
  7. Never pass complex objects — pass IDs only.

Likely Follow-up Questions

How do you share a ViewModel between screens in a flow?Scope the ViewModel to the nested navigation graph: viewModel(viewModelStoreOwner = navController.getBackStackEntry("checkoutGraph")). Both screens in the graph share the same instance; it's cleared when the graph is popped.
How do you handle back navigation from a composable?Use navController.navigateUp() (preferred — respects up navigation stack) or navController.popBackStack() (pops regardless). Pass the callback down as a lambda: onBack: () -> Unit = navController::navigateUp.
What's the difference between popBackStack and navigateUp?popBackStack() always pops the current destination. navigateUp() pops if there's a previous destination; otherwise delegates to the Activity's up navigation (useful for single-Activity with deep links).
How do deep links work in Navigation Compose?Declare navDeepLink { uriPattern = "https://..." } in composable(). Declare the intent-filter in AndroidManifest for the Activity. Navigation automatically routes the incoming Intent to the correct composable and extracts path parameters.

Senior-Level Insight

"The key advantage of Navigation Compose over Fragment Navigation is that there's no separate Fragment lifecycle — the composable back stack IS the navigation stack. There's no onCreateView/onDestroyView duality. State inside NavBackStackEntry persists across navigation (remember survives going back), and ViewModel scoping to a graph is cleaner than the Fragment activityViewModels() approach. The main trade-off is that Navigation Compose is less mature for very complex multi-module graphs — but Navigation 2.8's type-safe routes significantly close the gap, and I'd choose it for all new projects."

9 · Interview Questions (Self-Check)

Basic What are the three core components of Navigation Compose?
NavController (manages back stack state and provides navigate/popBackStack API), NavHost (Composable that renders the current destination from the back stack), NavGraph (the set of composable destinations and their routes, defined within the NavHost builder lambda).
Basic How do you pass an argument to a composable destination?
Declare the route with a placeholder: composable("product/{productId}", arguments = listOf(navArgument("productId") { type = NavType.StringType })) { backStackEntry -> val id = backStackEntry.arguments?.getString("productId") }. Navigate with: navController.navigate("product/abc123"). With Navigation 2.8+ typed routes: @Serializable data class ProductRoute(val productId: String), then navigate with navController.navigate(ProductRoute("abc123")) and val route: ProductRoute = backStackEntry.toRoute().
Basic How do you implement a logout flow that prevents the user from pressing Back to the home screen? 
navController.navigate(Route.Login) {
    popUpTo(0) { inclusive = true }  // clears entire back stack
}
popUpTo(0) removes all destinations back to the root (id=0). inclusive=true removes the start destination too. The user is left with only the Login screen and pressing Back exits the app.
Intermediate Explain the bottom navigation tab pattern with saveState and restoreState.
Without saveState/restoreState, switching tabs recreates the entire destination from scratch, losing scroll position and state. The pattern:
navController.navigate(tab.route) {
    popUpTo(navController.graph.findStartDestination().id) {
        saveState = true  // save the current tab's back stack
    }
    launchSingleTop = true  // don't stack duplicate
    restoreState = true     // restore previously saved state
}
This preserves each tab's individual state (scroll position, ViewModel state persists in memory) when switching between tabs.
Intermediate How do you scope a ViewModel to a nested navigation graph? 
// Define nested graph with a route
navigation(startDestination = "checkout_details", route = "checkout") {
    composable("checkout_details") { ... }
    composable("checkout_review") { ... }
    composable("checkout_payment") { ... }
}

// In any composable inside "checkout" graph:
val checkoutViewModel: CheckoutViewModel = viewModel(
    viewModelStoreOwner = remember(navController) {
        navController.getBackStackEntry("checkout")
    }
)
The ViewModel is created once and shared across all screens in the "checkout" graph. It's cleared when the entire graph is popped from the back stack.
Intermediate What is the difference between navigateUp() and popBackStack()?
popBackStack() always attempts to pop the current destination from the back stack, regardless of context. navigateUp() first tries to navigate up within the NavController's graph; if the back stack is empty, it delegates to the Activity's onSupportNavigateUp(), which handles system Up navigation (e.g., when the app was opened from a deep link and needs to go to a parent Activity). For in-app Back navigation, use navigateUp(). For programmatic navigation within flows, popBackStack() gives more control.
Intermediate How do you handle navigation events from a ViewModel safely?
Use a SharedFlow (or Channel as a Flow) to emit one-shot navigation events. Collect in the Composable with LaunchedEffect to ensure it only fires once even on recomposition:
// ViewModel
val navEvents = Channel<NavEvent>(Channel.BUFFERED)

// Composable
LaunchedEffect(Unit) {
    viewModel.navEvents.receiveAsFlow().collect { event ->
        when (event) {
            NavEvent.LoggedOut -> navController.navigate(Route.Login) {
                popUpTo(0) { inclusive = true }
            }
        }
    }
}
Using StateFlow for navigation events is a common mistake — it replays the last value on re-subscription, causing the navigation to fire again on recomposition.
Senior/Scenario [Scenario] After rotating the device mid-checkout (3 screens deep), the user is navigated back to the home screen. What went wrong?
The navigation event (e.g., navigate to home on auth state change) is being collected in a LaunchedEffect or from a StateFlow without lifecycle awareness, and fires again after rotation (because the state/event is still set). Specifically: if using StateFlow for a navigation event (instead of a one-shot Channel/SharedFlow), the last emitted value re-triggers on recomposition after rotation, causing an unintended navigation.

Fix: Use a Channel or SharedFlow(replay=0) for one-shot navigation events, not StateFlow. Alternatively, use ConsumedEvent wrappers or the official UiEvent pattern from Google's architecture samples. LaunchedEffect with a correct stable key (e.g., the event itself) ensures it only fires once per distinct event value.
Senior/Scenario [Coding] Implement a multi-module navigation setup where each feature module contributes its own NavGraph. 
// :core:navigation module
object AppRoutes {
    const val HOME_GRAPH = "home_graph"
    const val AUTH_GRAPH = "auth_graph"
}

// :feature:home module — extension function adds its destinations
fun NavGraphBuilder.homeGraph(navController: NavHostController) {
    navigation(startDestination = "home", route = AppRoutes.HOME_GRAPH) {
        composable("home") { HomeScreen(navController) }
        composable("product/{id}") { backStackEntry ->
            ProductScreen(backStackEntry.arguments?.getString("id")!!)
        }
    }
}

// :feature:auth module
fun NavGraphBuilder.authGraph(navController: NavHostController) {
    navigation(startDestination = "login", route = AppRoutes.AUTH_GRAPH) {
        composable("login") { LoginScreen(navController) }
        composable("register") { RegisterScreen(navController) }
    }
}

// :app module — assembles all graphs
@Composable
fun AppNavHost() {
    val navController = rememberNavController()
    NavHost(navController, startDestination = AppRoutes.HOME_GRAPH) {
        homeGraph(navController)
        authGraph(navController)
    }
}
Each feature module defines a NavGraphBuilder extension function. The :app module assembles them. No feature module depends on another feature module — they only depend on :core:navigation.
Senior/Scenario [Debug] Deep links from FCM notifications open the app but land on the wrong screen. What are the possible causes?
Common causes:
1. Wrong intent handling: The intent from the FCM PendingIntent is not being passed to the NavController. In a single-Activity setup, the NavController processes incoming Intents from handleDeepLink(intent) — this happens automatically if you use NavHost, but the Activity must call navController.handleDeepLink(intent) in onNewIntent().
2. Route mismatch: The URI pattern in navDeepLink{} doesn't match the URI in the notification exactly (trailing slash, query params, case sensitivity).
3. Missing intent-filter: The deep link intent-filter in AndroidManifest.xml doesn't match the URI scheme/host/path.
4. Auth state not ready: The NavHost starts with the unauthenticated start destination and navigates to login before the deep link is processed.
Fix: Add logging to navController.addOnDestinationChangedListener{} to trace every navigation. Verify URI patterns with adb: adb shell am start -W -a android.intent.action.VIEW -d "https://myapp.com/orders/123".
