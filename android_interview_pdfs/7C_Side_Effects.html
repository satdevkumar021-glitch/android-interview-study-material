1. Overview

What is a side effect in Compose?

A side effect is any operation that affects state or the outside world and is not directly part of describing the UI. Examples: launching a coroutine, registering a listener, calling analytics, updating external state. Because composables can re-run at any time, side effects must be carefully managed through dedicated APIs.

2. LaunchedEffect

LaunchedEffect(key) launches a coroutine when the composable enters composition. If the key changes, the current coroutine is cancelled and a new one is launched. When the composable leaves composition, the coroutine is cancelled.

// LaunchedEffect(Unit): run once when composable enters composition
@Composable
fun ProductScreen(productId: String, viewModel: ProductViewModel) {
    // Load when screen appears
    LaunchedEffect(Unit) {
        viewModel.loadProduct(productId)
    }

    // LaunchedEffect(productId): re-runs when productId changes
    LaunchedEffect(productId) {
        viewModel.loadProduct(productId)  // new productId → cancel old → start new
    }
}

// Snackbar: show when SnackbarEvent fires
@Composable
fun CheckoutScreen(viewModel: CheckoutViewModel) {
    val snackbarHostState = remember { SnackbarHostState() }

    LaunchedEffect(Unit) {
        viewModel.events.collect { event ->
            when (event) {
                is UiEvent.ShowSnackbar ->
                    snackbarHostState.showSnackbar(event.message)
                is UiEvent.Navigate ->
                    navController.navigate(event.route)
            }
        }
    }

    Scaffold(snackbarHost = { SnackbarHost(snackbarHostState) }) { /* ... */ }
}

// LaunchedEffect with multiple keys: re-runs when ANY key changes
LaunchedEffect(userId, filterType) {
    repo.loadFiltered(userId, filterType)
}

3. DisposableEffect

DisposableEffect(key) is for setup/teardown of non-suspending resources. It runs the block on composition and its onDispose { } block on disposal or key change.

// DisposableEffect: lifecycle observer registration
@Composable
fun LifecycleObserver(onResume: () -> Unit, onPause: () -> Unit) {
    val lifecycleOwner = LocalLifecycleOwner.current

    DisposableEffect(lifecycleOwner) {
        val observer = LifecycleEventObserver { _, event ->
            when (event) {
                Lifecycle.Event.ON_RESUME -> onResume()
                Lifecycle.Event.ON_PAUSE -> onPause()
                else -> {}
            }
        }
        lifecycleOwner.lifecycle.addObserver(observer)

        onDispose {
            lifecycleOwner.lifecycle.removeObserver(observer)  // ✅ cleanup
        }
    }
}

// DisposableEffect for MediaPlayer lifecycle
@Composable
fun VideoPlayer(url: String) {
    val context = LocalContext.current

    DisposableEffect(url) {        // re-creates on URL change
        val player = MediaPlayer()
        player.setDataSource(url)
        player.prepare()
        player.start()

        onDispose {
            player.stop()
            player.release()     // ✅ no resource leak
        }
    }
}

4. SideEffect & rememberCoroutineScope

SideEffect — every recomposition

// SideEffect: runs after every successful recomposition
// Use to sync Compose state to non-Compose systems
@Composable
fun AnalyticsScreen(screenName: String) {
    val analyticsTracker = LocalAnalyticsTracker.current

    SideEffect {
        // Runs after every recomposition — sync state to external system
        analyticsTracker.setScreen(screenName)
    }
}

// ❌ WRONG: do NOT use SideEffect for one-time operations
SideEffect { analytics.logScreenView() }  // logs on EVERY recomposition!
// ✅ Use LaunchedEffect(Unit) for one-time operations

rememberCoroutineScope — for event-triggered coroutines

// rememberCoroutineScope: a scope tied to composition lifetime
// Used when you need to launch a coroutine from a non-composable callback
@Composable
fun PayButton(onPay: suspend () -> Unit) {
    val scope = rememberCoroutineScope()
    var isLoading by remember { mutableStateOf(false) }

    Button(
        onClick = {                    // onClick is NOT a suspend function
            scope.launch {           // use scope to launch coroutine
                isLoading = true
                onPay()
                isLoading = false
            }
        },
        enabled = !isLoading
    ) {
        if (isLoading) CircularProgressIndicator()
        else Text("Pay")
    }
}
// When composable leaves, scope is cancelled → in-flight coroutines cancelled

5. produceState & snapshotFlow

produceState — convert non-Compose state to State<T>

// produceState: bridge non-Compose async sources to State<T>
@Composable
fun NetworkImage(url: String): State<ImageResult> = produceState(
    initialValue = ImageResult.Loading,
    key1 = url
) {
    value = imageLoader.load(url)   // suspend call
    // awaitDispose: cleanup if composable leaves before image loads
    awaitDispose { imageLoader.cancel(url) }
}

// Usage
@Composable
fun ProductImage(url: String) {
    val imageState by NetworkImage(url)
    when (imageState) {
        is ImageResult.Loading -> Shimmer()
        is ImageResult.Success -> Image(imageState.bitmap)
        is ImageResult.Error -> ErrorIcon()
    }
}

snapshotFlow — convert Compose State to Flow

// snapshotFlow: read Compose State inside a Flow context
// Emits when the read State values change
@Composable
fun AutoSaveForm(text: String, onSave: (String) -> Unit) {
    LaunchedEffect(Unit) {
        snapshotFlow { text }      // reads Compose state → becomes Flow
            .debounce(2000)         // wait 2s after user stops typing
            .distinctUntilChanged() // only if text actually changed
            .collect { onSave(it) }  // auto-save
    }
}

// snapshotFlow for analytics: track what user reads in a list
LaunchedEffect(Unit) {
    snapshotFlow { listState.firstVisibleItemIndex }
        .distinctUntilChanged()
        .collect { index -> analytics.trackItemImpression(items[index]) }
}

6. Pitfalls & Best Practices
Pitfall 1 — LaunchedEffect(Unit) for key-sensitive operations
Using LaunchedEffect(Unit) means the effect never re-runs when parameters change. If you load data by ID and the ID can change (navigation backstack, deep links), use LaunchedEffect(id). Forgetting to pass the correct key is a common source of stale data bugs.
Pitfall 2 — SideEffect for one-time operations
SideEffect runs on EVERY successful recomposition. Using it to log a screen view would fire the analytics event on every UI update, not just once. Use LaunchedEffect(Unit) for one-time effects.
Pitfall 3 — rememberCoroutineScope for one-time initialization
rememberCoroutineScope is for user-triggered events (button clicks). For automatic initialization on screen enter, use LaunchedEffect. Using rememberCoroutineScope + launching in the body (not a callback) will launch on every recomposition — same as calling the code directly.
Best Practice — DisposableEffect for any listener registration
Any time you register a listener, callback, or observer in a composable, always use DisposableEffect with onDispose cleanup. This prevents listener leaks when the composable leaves the composition.

7. Interview Framework

30-second answer

"Compose provides specific APIs for side effects. LaunchedEffect launches a coroutine tied to composition — re-runs when key changes, cancelled on leave. DisposableEffect is for non-suspend setup/teardown with onDispose cleanup. SideEffect runs after every recomposition for syncing external state. rememberCoroutineScope gives a scope for event-triggered coroutines. produceState converts async sources to State, snapshotFlow converts Compose State to Flow."

2-minute answer

"The key decision is: does the effect run once, on key change, on every recomposition, or on a user event? LaunchedEffect(key) handles the first two cases — it starts a coroutine when composition enters or key changes, and cancels it when leaving. The key is the dependency: LaunchedEffect(Unit) = run once; LaunchedEffect(userId) = re-run when user changes.

DisposableEffect handles non-coroutine resources: lifecycle observers, BroadcastReceivers, Bluetooth listeners. The onDispose block guarantees cleanup regardless of why the composable leaves.

SideEffect is specifically for syncing Compose state to non-Compose systems — like updating a legacy View system or a custom analytics SDK's current screen. It fires after every successful recomposition, ensuring external state stays in sync.

rememberCoroutineScope is used in click handlers and other non-composable callbacks where you need to launch a coroutine but aren't inside a suspend context."

Must-mention points

1
LaunchedEffect(key) — coroutine; re-runs on key change; cancel on leave
2
DisposableEffect + onDispose — non-suspend; cleanup guaranteed
3
SideEffect = every recomposition — not for one-time effects
4
rememberCoroutineScope = event-triggered — button clicks, not auto-init
5
snapshotFlow — converts Compose State to Flow; enables Flow operators on UI state

Follow-up Q&A

Q: "When would you use produceState vs LaunchedEffect + mutableStateOf?"

A: They're similar — produceState is cleaner syntactic sugar. produceState is preferred when the result is the entire return value of a composable (a reusable component that wraps an async source). LaunchedEffect + mutableStateOf is more flexible when you need multiple state updates inside the effect or more complex logic.

Q: "What is the key in LaunchedEffect — can it be anything?"

A: The key can be any value. Compose uses equals() to compare old and new key on recomposition. If they differ, the current coroutine is cancelled and a new one starts. Common keys: Unit (run once), a specific ID, a boolean state, a composable parameter. Multiple keys: LaunchedEffect(key1, key2) — re-runs if either changes.

Senior insight

Senior Insight
In a banking app, use snapshotFlow to convert the active account ID state to a Flow, then combine with server-sent events: LaunchedEffect(Unit) { snapshotFlow { activeAccountId }.flatMapLatest { id -> serverEvents.filterByAccount(id) }.collect { event -> handleLiveUpdate(event) } }. This gives you the full power of Flow operators on Compose state, including debounce, filtering, and flatMapLatest for account switching.

9. Interview Questions
What is LaunchedEffect and how does the key work?
Basic⌄
LaunchedEffect(key) launches a coroutine that is tied to the composition. When the composable enters composition, the coroutine starts. If the key changes on recomposition, the current coroutine is cancelled and a new one starts. When the composable leaves composition, the coroutine is cancelled. The key can be any value — Compose uses equals() to detect changes.
What is DisposableEffect and when must you use onDispose?
Basic⌄
DisposableEffect(key) runs a non-suspending setup block when the composable enters composition or when the key changes. The block must end with onDispose { } which runs when: the composable leaves composition, or the key changes (before re-running the setup block). onDispose is mandatory — it provides the cleanup for listeners, observers, resources registered in the setup block.
When should you use SideEffect vs LaunchedEffect?
Basic⌄
SideEffect runs after EVERY successful recomposition — use for continuously syncing Compose state to external non-Compose systems (e.g., updating an analytics SDK's current screen, updating a legacy View's property). LaunchedEffect(Unit) runs once when the composable enters composition — use for one-time initialization, loading data, starting observation. Never use SideEffect for one-time operations.
When would you use rememberCoroutineScope instead of LaunchedEffect?
Intermediate⌄
rememberCoroutineScope when you need to launch a coroutine in response to a user event (button click, gesture) — from a non-composable callback context where you can't call suspend functions directly. LaunchedEffect when the coroutine should start automatically based on composition entry or a state change. Rule: event-triggered → rememberCoroutineScope; automatic/reactive → LaunchedEffect.
A composable uses LaunchedEffect(Unit) to load data for a product. The user navigates to a different product. What happens and how do you fix it?
Intermediate⌄
With LaunchedEffect(Unit), the effect runs only once when the composable enters composition. If the same composable recomposes with a different productId, the effect does NOT re-run — the old data stays. Fix: LaunchedEffect(productId). Now when productId changes (e.g., navigation to a different product), the old coroutine is cancelled and a new one starts loading the new product.
How do you register and unregister a lifecycle observer safely in Compose?
Intermediate⌄
Use DisposableEffect(lifecycleOwner): in the block, create and add the observer; in onDispose, remove it. Use LocalLifecycleOwner.current to get the current lifecycle owner and pass it as the key — if the owner changes (e.g., nested Navigation), the effect re-runs. Without DisposableEffect (e.g., registering directly in the body), the observer would be added on every recomposition and never removed — a leak.
What is snapshotFlow and what can you do with it that you can't do with just collecting a StateFlow?
Intermediate⌄
snapshotFlow { } converts any Compose State read inside its lambda into a Flow. Unlike a StateFlow (which has a value), snapshotFlow can read any combination of State values: snapshotFlow { state1 + state2 } — emits when either state1 or state2 changes. You can then apply all Flow operators: debounce, distinctUntilChanged, flatMapLatest. This enables complex reactive patterns on local Compose state that wouldn't be possible with just StateFlow.
Design an auto-save feature for a notes app that saves to the database after the user stops typing for 2 seconds.
Senior⌄
@Composable fun NoteEditor(noteId: String, viewModel: NoteViewModel) { val content by viewModel.content.collectAsStateWithLifecycle(); LaunchedEffect(Unit) { snapshotFlow { content }.debounce(2000).distinctUntilChanged().drop(1) // skip initial load .collect { text -> viewModel.saveNote(noteId, text) } }; TextField(value = content, onValueChange = viewModel::onContentChange) }. snapshotFlow converts the content state to Flow. debounce(2000) waits 2s after last keystroke. distinctUntilChanged prevents saving if content didn't change. drop(1) skips the initial emission when the composable enters (don't save before user types).
What is produceState and how is it different from LaunchedEffect + mutableStateOf?
Senior⌄
produceState(initialValue, key) { value = compute() } is syntactic sugar for LaunchedEffect + mutableStateOf — internally it does exactly that. The difference is idiomatic: produceState returns a State<T> directly, making it ideal for composable functions that compute and return a State (not just a void composable). It also provides awaitDispose { } for cleanup. Use produceState when building a composable abstraction over an async data source; use LaunchedEffect when the coroutine is a side effect of an existing composable, not its primary purpose.
A developer registers a BroadcastReceiver in a composable body (not in DisposableEffect). What bugs will occur?
Senior⌄
Three bugs: (1) The receiver is registered on EVERY recomposition — if the screen recomposes 10 times, 10 receivers are registered. Each will respond to broadcasts, causing the handler to fire 10x. (2) No unregistration ever happens — when the composable leaves composition, no cleanup runs. The receiver continues receiving broadcasts forever, holding a reference to the composable's context. (3) Memory leak — the receiver holds a context reference; if the context is Activity-scoped, the Activity leaks. Fix: DisposableEffect with onDispose { context.unregisterReceiver(receiver) }.
