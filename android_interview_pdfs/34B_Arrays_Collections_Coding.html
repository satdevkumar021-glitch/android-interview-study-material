2. Two Sum
1
Find indices of two numbers that add to target. [2,7,11,15], target=9 → [0,1]
O(n) time · O(n) space
Brute Force O(n²)
fun twoSumBrute(nums: IntArray, target: Int): IntArray {
    for (i in nums.indices) {
        for (j in i + 1 until nums.size) {
            if (nums[i] + nums[j] == target) return intArrayOf(i, j)
        }
    }
    return intArrayOf()  // no solution found
}
Optimal: HashMap O(n)
fun twoSum(nums: IntArray, target: Int): IntArray {
    val seen = HashMap<Int, Int>()   // value → index

    for ((index, num) in nums.withIndex()) {
        val complement = target - num    // what we need to find
        if (complement in seen) {
            return intArrayOf(seen[complement]!!, index)
        }
        seen[num] = index
    }
    return intArrayOf()
}

// Trace: nums=[2,7,11,15], target=9
// i=0: complement=7, not in seen. seen={2:0}
// i=1: complement=2, in seen at index 0! → return [0,1]

println(twoSum(intArrayOf(2,7,11,15), 9).contentToString())  // [0, 1]

3. Remove Duplicates from Sorted Array
2
[1,1,2,3,3,4] → [1,2,3,4], return length 4 (in-place)
O(n) time · O(1) space
Two-pointer in-place approach
fun removeDuplicates(nums: IntArray): Int {
    if (nums.isEmpty()) return 0

    var slow = 0   // pointer to last unique position
    for (fast in 1 until nums.size) {
        if (nums[fast] != nums[slow]) {
            slow++
            nums[slow] = nums[fast]  // place next unique element
        }
    }
    return slow + 1  // length of unique elements
}

// Trace: [1,1,2,3,3,4]
// slow=0, fast=1: 1==1, skip
// slow=0, fast=2: 1!=2, slow=1, nums[1]=2 → [1,2,2,3,3,4]
// slow=1, fast=3: 2!=3, slow=2, nums[2]=3 → [1,2,3,3,3,4]
// slow=2, fast=4: 3==3, skip
// slow=2, fast=5: 3!=4, slow=3, nums[3]=4 → [1,2,3,4,3,4]
// return 4

// Kotlin idiomatic (not in-place but clean):
fun removeDuplicatesKt(nums: IntArray): IntArray = nums.toSortedSet().toIntArray()

3
Sort and group elements by multiple criteria
O(n log n) · O(n) space
Sort by multiple criteria
data class Product(val name: String, val price: Double, val category: String)

val products = listOf(
    Product("Apple", 1.5, "Fruit"),
    Product("Banana", 0.8, "Fruit"),
    Product("Broccoli", 2.0, "Veg")
)

// Sort by category ascending, then by price ascending
val sorted = products.sortedWith(compareBy({ it.category }, { it.price }))

// Sort by price descending
val byPriceDesc = products.sortedByDescending { it.price }

// Sort with natural order then custom comparator
val custom = products.sortedWith(compareBy<Product> { it.category }.thenByDescending { it.price })
Group elements
// groupBy: group into Map<Key, List<T>>
val byCategory: Map<String, List<Product>> = products.groupBy { it.category }
// {"Fruit": [Apple, Banana], "Veg": [Broccoli]}

// groupingBy + eachCount: count occurrences
val words = listOf("apple", "banana", "apple", "cherry", "banana", "banana")
val wordCount: Map<String, Int> = words.groupingBy { it }.eachCount()
// {"apple":2, "banana":3, "cherry":1}

// Count and find most common word
val mostCommon = wordCount.maxByOrNull { it.value }?.key
// "banana"

// partition: split into two lists based on predicate
val (expensive, cheap) = products.partition { it.price > 1.0 }

5. Kadane's Algorithm — Maximum Subarray
4
[-2,1,-3,4,-1,2,1,-5,4] → 6 (subarray [4,-1,2,1])
O(n) time · O(1) space
Brute Force O(n²)
fun maxSubarrayBrute(nums: IntArray): Int {
    var max = Int.MIN_VALUE
    for (i in nums.indices) {
        var sum = 0
        for (j in i until nums.size) {
            sum += nums[j]
            max = maxOf(max, sum)
        }
    }
    return max
}
Kadane's Algorithm O(n)
fun maxSubarray(nums: IntArray): Int {
    var currentSum = nums[0]
    var maxSum = nums[0]

    for (i in 1 until nums.size) {
        // Either extend the previous subarray, or start fresh from nums[i]
        currentSum = maxOf(nums[i], currentSum + nums[i])
        maxSum = maxOf(maxSum, currentSum)
    }
    return maxSum
}

// Trace: [-2,1,-3,4,-1,2,1,-5,4]
// i=1: current = max(1, -2+1=-1) = 1, max = 1
// i=2: current = max(-3, 1-3=-2) = -2, max = 1
// i=3: current = max(4, -2+4=2) = 4, max = 4
// i=4: current = max(-1, 4-1=3) = 3, max = 4
// i=5: current = max(2, 3+2=5) = 5, max = 5
// i=6: current = max(1, 5+1=6) = 6, max = 6
// i=7: current = max(-5, 6-5=1) = 1, max = 6
// i=8: current = max(4, 1+4=5) = 5, max = 6 → return 6

println(maxSubarray(intArrayOf(-2,1,-3,4,-1,2,1,-5,4)))  // 6
Kadane's Insight
The key decision at each element: "Is it better to extend the current subarray (currentSum + nums[i]) or start fresh (nums[i])?" Start fresh when the current sum has become negative — a negative sum is worse than starting over.

Rotate Array

// Rotate array k positions to the right: [1,2,3,4,5], k=2 → [4,5,1,2,3]
fun rotate(nums: IntArray, k: Int) {
    val n = nums.size
    val shift = k % n   // handle k > n
    val rotated = nums.takeLast(shift) + nums.take(n - shift)
    rotated.forEachIndexed { i, v -> nums[i] = v }
}

// Or in-place with three reverses:
fun rotateInPlace(nums: IntArray, k: Int) {
    val n = nums.size; val shift = k % n
    fun reverse(a: Int, b: Int) { var l=a; var r=b; while(l<r){val t=nums[l];nums[l]=nums[r];nums[r]=t;l++;r--} }
    reverse(0, n-1); reverse(0, shift-1); reverse(shift, n-1)
}

6. HashMap Problems
5
Word frequency count and most common element
O(n) time · O(n) space
Word frequency from text
fun wordFrequency(text: String): Map<String, Int> =
    text.lowercase()
        .split("\\s+".toRegex())
        .filter { it.isNotEmpty() }
        .groupingBy { it }
        .eachCount()

val text = "the quick brown fox jumps over the lazy dog the"
val freq = wordFrequency(text)
println(freq["the"])  // 3

// Top 3 most frequent words
val top3 = freq.entries.sortedByDescending { it.value }.take(3)
    .map { it.key }
Most common element in array
fun mostCommon(nums: IntArray): Int? {
    val freq = HashMap<Int, Int>()
    for (n in nums) freq[n] = (freq[n] ?: 0) + 1
    return freq.maxByOrNull { it.value }?.key
}

// Kotlin idiomatic
fun mostCommonKt(nums: IntArray): Int? =
    nums.toList().groupingBy { it }.eachCount().maxByOrNull { it.value }?.key

println(mostCommon(intArrayOf(1,3,2,3,1,3)))  // 3

// Two Sum variation: count pairs with difference k
fun countPairsWithDiff(nums: IntArray, k: Int): Int {
    val set = nums.toHashSet()
    return nums.count { (it + k) in set }
}

9. Practice Questions
Explain the HashMap trick that reduces Two Sum from O(n²) to O(n).
Easy▼
Brute force checks every pair (i, j) — O(n²). The insight: for each element x, we need to find (target - x) somewhere in the array. Instead of searching through the array again (O(n) per element), store each element in a HashMap as we go. When we encounter x, look up (target - x) in the HashMap in O(1). Total: one pass = O(n) time. We're trading space O(n) for time.
What is Kadane's algorithm and why does it work?
Medium▼
Kadane's algorithm finds the maximum sum contiguous subarray in O(n) time. At each element, it makes a greedy choice: should I extend the current subarray (currentSum + nums[i]) or start a new one (nums[i])? If currentSum is negative, starting fresh is always better — a negative prefix only drags down the sum. currentSum = max(nums[i], currentSum + nums[i]). Track maxSum = max(maxSum, currentSum) throughout. This works because the optimal subarray ends at some index and starts at the best starting point.
Write code to find all pairs in an array that sum to a target value.
Medium▼
fun findPairs(nums: IntArray, target: Int): List<Pair<Int,Int>> { val seen = HashSet<Int>(); val result = mutableListOf<Pair<Int,Int>>(); val added = HashSet<Int>(); for (n in nums) { val complement = target - n; if (complement in seen && n !in added) { result.add(n to complement); added.add(n); added.add(complement) }; seen.add(n) }; return result }. The added set prevents duplicate pairs.
How do you sort a list of data class instances by multiple fields in Kotlin?
Easy▼
list.sortedWith(compareBy({ it.field1 }, { it.field2 })) — sorts by field1 first, then by field2 for ties. For descending on one field: compareBy<T> { it.field1 }.thenByDescending { it.field2 }. Alternatively: list.sortedWith(compareByDescending { it.price }) for single descending. The lambda parameters are evaluated in order — first lambda is primary sort key, second is tiebreaker.
What is the difference between groupBy and groupingBy in Kotlin?
Medium▼
groupBy { key } immediately creates a Map<K, List<V>> — all elements grouped into lists. It's eager and materializes the entire result. groupingBy { key } creates a lazy Grouping object — you then chain operations: eachCount() (count per group), fold() (custom reduction), aggregate() (custom). groupingBy is more memory-efficient for operations that don't need the full lists, like counting. eachCount() is groupingBy's most common use.
Write code to find the intersection of two arrays (common elements).
Easy▼
fun intersection(a: IntArray, b: IntArray): IntArray { val setA = a.toHashSet(); return b.filter { it in setA }.toIntArray() }. O(n + m) time. For distinct results: b.distinct().filter { it in setA }. Kotlin idiomatic: (a.toSet() intersect b.toSet()).toIntArray(). The intersect extension function on sets directly computes set intersection.
How does the two-pointer approach remove duplicates from a sorted array in O(1) space?
Medium▼
Use two pointers: slow (last unique position) and fast (scanner). Fast advances every step. When nums[fast] != nums[slow], we found a new unique element: slow++, copy nums[fast] to nums[slow]. Since the array is sorted, duplicates are adjacent — when fast encounters a new value, it's guaranteed to be unique. This overwrites duplicates with later unique values. No extra array needed → O(1) space.
Write a function to find the majority element (appears more than n/2 times).
Hard▼
Boyer-Moore Voting Algorithm — O(n) time, O(1) space: fun majorityElement(nums: IntArray): Int { var candidate = nums[0]; var count = 1; for (i in 1 until nums.size) { if (count == 0) { candidate = nums[i]; count = 1 } else if (nums[i] == candidate) count++ else count-- }; return candidate }. Intuition: majority element "survives" against all others in a vote — its count never reaches 0 if it truly appears more than n/2 times.
How would you merge two sorted arrays into one sorted array?
Easy▼
Two-pointer approach: var i=0, j=0; result = IntArray(a.size + b.size); for each position: if a[i] ≤ b[j], take a[i], else take b[j]; after one array exhausted, copy remaining. O(n+m) time, O(n+m) space. Kotlin one-liner: (a.toList() + b.toList()).sorted().toIntArray() — O((n+m) log(n+m)) but simpler. The two-pointer approach is O(n+m) and is the optimal solution to know for interviews.
What does the partition function do in Kotlin collections?
Easy▼
partition { predicate } splits a collection into two lists: (matching, notMatching). Returns a Pair<List<T>, List<T>>. Example: val (evens, odds) = listOf(1,2,3,4,5).partition { it % 2 == 0 }. Destructuring the Pair into named variables makes it very readable. Under the hood, it's O(n) — iterates once and puts each element in one of two lists. Useful for splitting positive/negative, active/inactive, etc.
